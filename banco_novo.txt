CREATE TABLE USUARIO (
    ID_USUARIO INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR(150) NOT NULL,
    EMAIL VARCHAR(150) NOT NULL UNIQUE,
    SENHA VARCHAR(255) NOT NULL,

    -- NOVOS CAMPOS NECESSÁRIOS PARA O FLASK
    ID_TIPO INTEGER NOT NULL,
    TIPO_NOME VARCHAR(20) NOT NULL,

    CONTA_CONFIRMADA BOOLEAN DEFAULT FALSE,
    BLOQUEADO BOOLEAN DEFAULT FALSE,
    TENTATIVAS_LOGIN INTEGER DEFAULT 0,

    DATA_CRIACAO TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    ATIVO BOOLEAN DEFAULT TRUE
);




CREATE TABLE HISTORICO_SENHAS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    SENHA_HASH VARCHAR(255) NOT NULL,

    CONSTRAINT FK_HIST_SENHA_USUARIO
        FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO(ID_USUARIO)
        ON DELETE CASCADE
);




CREATE TABLE CODIGOS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    CODIGO VARCHAR(10) NOT NULL,
    TIPO VARCHAR(30) NOT NULL,
    UTILIZADO BOOLEAN DEFAULT FALSE,

    CONSTRAINT FK_CODIGO_USUARIO
        FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO(ID_USUARIO)
        ON DELETE CASCADE
);




CREATE TABLE RECUPERAR_SENHA (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    CODIGO VARCHAR(10) NOT NULL,
    EXPIRACAO TIMESTAMP NOT NULL,
    UTILIZADO BOOLEAN DEFAULT FALSE,

    CONSTRAINT FK_RECUPERAR_USUARIO
        FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO(ID_USUARIO)
        ON DELETE CASCADE
);




CREATE TABLE PRODUTOS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR(150) NOT NULL,
    DESCRICAO BLOB SUB_TYPE TEXT,
    PRECO DECIMAL(10,2) NOT NULL,
    ATIVO BOOLEAN DEFAULT TRUE
);




CREATE TABLE MESAS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NUMERO INTEGER NOT NULL,
    STATUS VARCHAR(20) NOT NULL,
    DATA_ABERTURA DATE,
    HORA_ABERTURA TIME
);




CREATE TABLE PEDIDOS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    MESA_ID INTEGER,

    DATA_PEDIDO DATE NOT NULL,
    HORA_PEDIDO TIME NOT NULL,

    STATUS VARCHAR(20) NOT NULL,

    CONSTRAINT FK_PEDIDO_MESA
        FOREIGN KEY (MESA_ID)
        REFERENCES MESAS(ID)
);




CREATE TABLE ITENSPEDIDO (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    PEDIDO_ID INTEGER NOT NULL,
    PRODUTO_ID INTEGER NOT NULL,

    QUANTIDADE INTEGER NOT NULL,
    PRECO_UNITARIO DECIMAL(10,2) NOT NULL,
    SUBTOTAL DECIMAL(10,2) NOT NULL,

    CONSTRAINT FK_ITEM_PEDIDO
        FOREIGN KEY (PEDIDO_ID)
        REFERENCES PEDIDOS(ID)
        ON DELETE CASCADE,

    CONSTRAINT FK_ITEM_PRODUTO
        FOREIGN KEY (PRODUTO_ID)
        REFERENCES PRODUTOS(ID)
);




CREATE TABLE PAGAMENTOS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    MESA_ID INTEGER,

    VALOR_TOTAL DECIMAL(10,2) NOT NULL,
    FORMA_PAGAMENTO VARCHAR(30) NOT NULL,

    DATA_PAGAMENTO DATE NOT NULL,
    HORA_PAGAMENTO TIME NOT NULL,

    CONSTRAINT FK_PAGAMENTO_MESA
        FOREIGN KEY (MESA_ID)
        REFERENCES MESAS(ID)
);




CREATE TABLE HISTORICO_MESAS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    MESA_ID INTEGER NOT NULL,

    DATA_ABERTURA DATE NOT NULL,
    HORA_ABERTURA TIME NOT NULL,

    DATA_FECHAMENTO DATE,
    HORA_FECHAMENTO TIME,

    VALOR_TOTAL DECIMAL(10,2),

    ITENS_CONSUMIDOS BLOB SUB_TYPE TEXT,

    CONSTRAINT FK_HISTORICO_MESA
        FOREIGN KEY (MESA_ID)
        REFERENCES MESAS(ID)
);




CREATE TABLE TRANSFERENCIAS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    USUARIO_ORIGEM_ID INTEGER NOT NULL,
    USUARIO_DESTINO_ID INTEGER NOT NULL,

    VALOR DECIMAL(10,2) NOT NULL,
    DESCRICAO VARCHAR(255),

    DATA_TRANSFERENCIA TIMESTAMP NOT NULL,

    STATUS VARCHAR(20) NOT NULL,

    CONSTRAINT FK_TRANSF_USUARIO_ORIGEM
        FOREIGN KEY (USUARIO_ORIGEM_ID)
        REFERENCES USUARIO(ID_USUARIO),

    CONSTRAINT FK_TRANSF_USUARIO_DESTINO
        FOREIGN KEY (USUARIO_DESTINO_ID)
        REFERENCES USUARIO(ID_USUARIO)
);



CREATE TABLE TRANSFERENCIA_ITENS (
    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    ITEM_PEDIDO_ID INTEGER NOT NULL,

    MESA_ORIGEM_ID INTEGER NOT NULL,
    MESA_DESTINO_ID INTEGER NOT NULL,

    DATA_TRANSFERENCIA DATE NOT NULL,
    HORA_TRANSFERENCIA TIME NOT NULL,

    CONSTRAINT FK_TRANS_ITEM
        FOREIGN KEY (ITEM_PEDIDO_ID)
        REFERENCES ITENSPEDIDO(ID),

    CONSTRAINT FK_TRANS_MESA_ORIGEM
        FOREIGN KEY (MESA_ORIGEM_ID)
        REFERENCES MESAS(ID),

    CONSTRAINT FK_TRANS_MESA_DESTINO
        FOREIGN KEY (MESA_DESTINO_ID)
        REFERENCES MESAS(ID)
);

DROP TABLE TRANSFERENCIA_ITENS;

DROP TABLE TRANSFERENCIAS;

DROP TABLE PAGAMENTOS;

DROP TABLE ITENSPEDIDO;

DROP TABLE PEDIDOS;

DROP TABLE MESAS;

DROP TABLE PRODUTOS;

DROP TABLE RECUPERACAO_SENHA;

DROP TABLE CODIGOS;

DROP TABLE HISTORICO_SENHAS;

DROP TABLE USUARIO;

    ID INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    CODIGO VARCHAR(10) NOT NULL,
    EXPIRACAO TIMESTAMP NOT NULL,
    UTILIZADO BOOLEAN DEFAULT FALSE,
    CONSTRAINT FK_RECUPERAR_USUARIO FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO) ON DELETE CASCADE
